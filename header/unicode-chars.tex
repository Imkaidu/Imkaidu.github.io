% Unicode character fallbacks for pdflatex, shared by both the PDF and
% slides builds (script/generate.sh) so a character mapped here does not

% because that file also loads fancyhdr/pagestyle content that beamer
% does not want.

\newunicodechar{●}{\textbullet}
\newunicodechar{├}{\textbar}
\newunicodechar{└}{\textbar}
\newunicodechar{─}{-}
\newunicodechar{→}{\ensuremath{\rightarrow}}
